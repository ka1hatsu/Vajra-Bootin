cmake_minimum_required(VERSION 3.24)
project(VajraBootin VERSION 0.6.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Vajra Bootin currently targets Windows only.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(FetchContent)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.92.5 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(imgui)
add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC d3d11 dxgi)
add_library(vajra_recommender STATIC src/catalog/Distro.cpp src/catalog/ReleaseArtifact.cpp src/catalog/ChecksumManifest.cpp src/catalog/ArtifactCache.cpp src/recommender/Recommender.cpp)
target_include_directories(vajra_recommender PUBLIC src)
target_compile_options(vajra_recommender PRIVATE /W4 /permissive- /EHsc)
add_library(vajra_download STATIC src/download/SourcePolicy.cpp src/download/DownloadManager.cpp src/download/MetadataClient.cpp)
target_include_directories(vajra_download PUBLIC src)
target_compile_options(vajra_download PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(vajra_download PUBLIC winhttp)
add_library(vajra_verification STATIC src/verification/Sha256.cpp)
target_include_directories(vajra_verification PUBLIC src)
target_compile_options(vajra_verification PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(vajra_verification PUBLIC bcrypt)
add_library(vajra_workflow STATIC src/workflow/VerifiedDownload.cpp src/workflow/DownloadSelection.cpp src/workflow/ArtifactResolver.cpp src/workflow/UbuntuReleaseService.cpp)
target_include_directories(vajra_workflow PUBLIC src)
target_compile_options(vajra_workflow PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(vajra_workflow PUBLIC vajra_download vajra_verification vajra_recommender)
add_executable(Vajra WIN32 src/main.cpp src/app/Application.cpp src/hardware/Scanner.cpp src/ui/AppUi.cpp)
target_include_directories(Vajra PRIVATE src)
target_compile_definitions(Vajra PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Vajra PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(Vajra PRIVATE imgui vajra_recommender vajra_workflow d3d11 dxgi advapi32)
if(BUILD_TESTING)
    function(vajra_test target source library test_name)
        add_executable(${target} ${source})
        target_link_libraries(${target} PRIVATE ${library})
        target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
        add_test(NAME ${test_name} COMMAND ${target})
    endfunction()
    vajra_test(VajraRecommenderTests tests/RecommenderTests.cpp vajra_recommender recommender_rules)
    vajra_test(VajraReleaseArtifactTests tests/ReleaseArtifactTests.cpp vajra_recommender release_artifact_catalog)
    vajra_test(VajraChecksumManifestTests tests/ChecksumManifestTests.cpp vajra_recommender checksum_manifest_parser)
    vajra_test(VajraArtifactCacheTests tests/ArtifactCacheTests.cpp vajra_recommender artifact_cache_validation)
    vajra_test(VajraSourcePolicyTests tests/SourcePolicyTests.cpp vajra_download trusted_source_policy)
    vajra_test(VajraDownloadManagerTests tests/DownloadManagerTests.cpp vajra_download download_manager_guards)
    vajra_test(VajraSha256Tests tests/Sha256Tests.cpp vajra_verification sha256_verification)
    vajra_test(VajraVerifiedDownloadTests tests/VerifiedDownloadTests.cpp vajra_workflow verified_download_workflow)
    vajra_test(VajraDownloadSelectionTests tests/DownloadSelectionTests.cpp vajra_workflow recommendation_download_handoff)
    vajra_test(VajraArtifactResolverTests tests/ArtifactResolverTests.cpp vajra_workflow publisher_artifact_resolver)
    vajra_test(VajraUbuntuReleaseServiceTests tests/UbuntuReleaseServiceTests.cpp vajra_workflow ubuntu_live_release_parser)
endif()
set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT Vajra)
